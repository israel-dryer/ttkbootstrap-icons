# Documentation build dependencies.
#
# Deliberately not an extra of the base package. `pip install tkinter-icons`
# should pull a renderer and Pillow, not a documentation toolchain, and the
# family (bootstack, ttkbootstrap) keeps its docs requirements in this file.
#
# Building the docs also needs the packs installed, because the packs page
# reports each set's real style list, upstream version, and glyph count by
# importing its provider. See docs/conf.py.
sphinx>=7.2
pydata-sphinx-theme
sphinx-design
sphinx-copybutton
sphinx-autobuild
myst-parser
# PySimpleGUI is not a dependency of the library — it is the optional framework
# `tkinter_icons.extensions.psg` bridges onto. It is here because `api.rst`
# autodocs `IconButton`, and `IconButton` subclasses PySimpleGUI's own `Button`,
# so autodoc has to import it. Without this the docs build fails on every
# machine that has not installed it, including Read the Docs.
PySimpleGUI
